<script>
(() => {
  const FEED = 'https://sti-keeper-production.up.railway.app';
  const CAMPAIGNS = 'https://sti-campaigns-production.up.railway.app';
  const SITE = 'https://sti.boombots.fun';
  const q = new URLSearchParams(location.search);
  const want = (q.get('coin') || '').trim();
  const strip = q.get('style') === 'strip';
  const theme = q.get('theme') || 'dark';
  const light = theme === 'light' || (theme === 'auto' && matchMedia('(prefers-color-scheme: light)').matches);
  const root = document.getElementById('badge');
  if (light) document.documentElement.classList.add('light');

  const el = (tag, cls, text) => {
    const n = document.createElement(tag);
    if (cls) n.className = cls;
    if (text !== undefined) n.textContent = text;
    return n;
  };
  const norm = (t) => t.toLowerCase().replace(/^0x0*/, '0x');
  const key = (s) => s.toUpperCase().replace(/[^A-Z0-9]/g, '');
  const compact = (raw, decimals) => {
    const n = Number(raw) / 10 ** decimals;
    const [d, u] = n >= 1e9 ? [1e9, 'B'] : n >= 1e6 ? [1e6, 'M'] : n >= 1e3 ? [1e3, 'K'] : [1, ''];
    const v = n / d;
    return `${v >= 100 ? Math.round(v) : v >= 10 ? v.toFixed(1) : v.toFixed(2)}${u}`;
  };
  const percent = (x) => `${(x * 100).toFixed(x < 0.1 ? 1 : 0)}%`;

  // ── the accent: the logo's own colour ──────────────────────────────────
  const hsl = (h, s, l) => `hsl(${Math.round(h)} ${Math.round(s * 100)}% ${Math.round(l * 100)}%)`;
  const BG = light ? [255, 255, 255] : [14, 42, 51];
  function toRgb(h, s, l) {
    const k = (n) => (n + h / 30) % 12, a = s * Math.min(l, 1 - l);
    return [0, 8, 4].map((n) => 255 * (l - a * Math.max(-1, Math.min(k(n) - 3, 9 - k(n), 1))));
  }
  const lum = (rgb) => rgb.map((v) => { v /= 255; return v <= 0.03928 ? v / 12.92 : ((v + 0.055) / 1.055) ** 2.4; })
    .reduce((a, v, i) => a + v * [0.2126, 0.7152, 0.0722][i], 0);
  const contrast = (a, b) => (Math.max(lum(a), lum(b)) + 0.05) / (Math.min(lum(a), lum(b)) + 0.05);
  /**
   * The accent from a logo's hue. The fill keeps the brand's own lightness
   * within reason; the text colour is walked lighter (or darker, on the light
   * theme) until it reads on the card: a deep blue logo still gets legible numbers.
   */
  function setAccent(h, s, l) {
    const S = Math.max(s, 0.5);
    const fill = Math.min(Math.max(l, light ? 0.36 : 0.5), light ? 0.52 : 0.68);
    let ink = fill;
    while (contrast(toRgb(h, S, ink), BG) < 4.5 && ink > 0.05 && ink < 0.95) ink += light ? -0.02 : 0.02;
    const st = document.documentElement.style;
    st.setProperty('--accent', hsl(h, S, fill));
    st.setProperty('--accent-ink', hsl(h, S, ink));
    st.setProperty('--on-accent', contrast(toRgb(h, S, fill), [255, 255, 255]) >= contrast(toRgb(h, S, fill), [14, 42, 51]) ? '#FFFFFF' : '#0E2A33');
  }
  function toHsl(r, g, b) {
    r /= 255; g /= 255; b /= 255;
    const max = Math.max(r, g, b), min = Math.min(r, g, b), l = (max + min) / 2, d = max - min;
    if (!d) return [0, 0, l];
    const s = d / (1 - Math.abs(2 * l - 1));
    const h = max === r ? ((g - b) / d + 6) % 6 : max === g ? (b - r) / d + 2 : (r - g) / d + 4;
    return [h * 60, s, l];
  }
  /** The logo's dominant colour: pixels vote by hue, the vivid ones count most. */
  function accentFrom(img) {
    try {
      const c = document.createElement('canvas');
      c.width = c.height = 32;
      const ctx = c.getContext('2d', { willReadFrequently: true });
      ctx.drawImage(img, 0, 0, 32, 32);
      const px = ctx.getImageData(0, 0, 32, 32).data;
      const bins = Array.from({ length: 18 }, () => ({ w: 0, x: 0, y: 0, s: 0, l: 0 }));
      for (let i = 0; i < px.length; i += 4) {
        if (px[i + 3] < 200) continue;
        const [h, s, l] = toHsl(px[i], px[i + 1], px[i + 2]);
        if (s < 0.25 || l < 0.15 || l > 0.9) continue;
        const w = s * (1 - Math.abs(l - 0.5));
        const b = bins[Math.floor(h / 20) % 18];
        b.w += w; b.s += s * w; b.l += l * w;
        b.x += Math.cos(h * Math.PI / 180) * w; b.y += Math.sin(h * Math.PI / 180) * w;
      }
      const best = bins.reduce((a, b) => (b.w > a.w ? b : a));
      if (best.w < 8) return;                       // a black-and-white logo keeps the index's amber
      setAccent((Math.atan2(best.y, best.x) * 180 / Math.PI + 360) % 360, best.s / best.w, best.l / best.w);
    } catch { /* keep the default */ }
  }
  const given = (q.get('accent') || '').replace('#', '');
  if (/^[0-9a-f]{6}$/i.test(given)) {
    setAccent(...toHsl(parseInt(given.slice(0, 2), 16), parseInt(given.slice(2, 4), 16), parseInt(given.slice(4, 6), 16)));
  }

  function logo(coin) {
    const letter = () => el('div', 'logo letter', coin.symbol.slice(0, 1));
    if (!coin.icon) return letter();
    const img = el('img', 'logo');
    const src = typeof coin.icon === 'string' ? coin.icon : `${FEED}/icon/${coin.key}`;
    img.alt = '';
    img.crossOrigin = 'anonymous';
    img.onload = () => { if (!given) accentFrom(img); };
    img.onerror = () => img.replaceWith(letter());
    img.src = src;
    return img;
  }

  function core(feed, coin) {
    const bar = el('div', 'core');
    const parts = feed.coins.map((c) => ({ c, w: c.share ?? c.targetBps / 10_000 })).sort((a, b) => b.w - a.w);
    for (const p of parts) {
      const i = el('i', coin && p.c.key === coin.key ? 'me' : '');
      i.style.flex = `${Math.max(p.w, 0.002) * 1000} 1 0`;
      bar.append(i);
    }
    return bar;
  }

  function place() {
    const me = root.querySelector('.card .core .me');
    const tick = root.querySelector('.tick');
    if (!me || !tick) return;
    const x = me.offsetLeft - me.parentElement.offsetLeft + me.offsetWidth / 2;
    const span = tick.firstElementChild;
    tick.style.setProperty('--x', `${x}px`);
    const w = span.offsetWidth, room = tick.offsetWidth;
    span.style.left = `${Math.min(Math.max(x - 10, 0), Math.max(room - w, 0))}px`;
  }

  /** A long symbol shrinks to fit rather than being cut off. */
  function fit() {
    const sym = root.querySelector('.card .sym');
    if (!sym) return;
    let size = 22;
    sym.style.fontSize = '';
    while (sym.scrollWidth > sym.clientWidth && size > 13) sym.style.fontSize = `${--size}px`;
  }

  function tell() {
    if (parent === window) return;
    parent.postMessage({ sti: 'badge-size', id: q.get('id'), height: Math.ceil(root.getBoundingClientRect().height) }, '*');
  }

  /** The card for a coin that is campaigning for a place. c: its row from the campaigns service. */
  function renderCampaign(feed, c, all) {
    const active = all.filter((x) => x.rank).length;
    const ahead = all.find((x) => x.rank === c.rank - 1);
    const standing = !c.rank ? 'No votes this week. One vote puts it back in the race.'
      : c.rank === 1 ? (active > 1 ? `Leading ${active} campaigns` : 'Leading the race')
        : `#${c.rank} of ${active} · ${c.behind.toLocaleString()} behind ${ahead ? ahead.symbol : 'the next'}`;
    root.replaceChildren();
    root.className = strip ? 'strip' : 'card';
    root.href = `${SITE}/#campaign-${encodeURIComponent(c.key)}`;
    root.setAttribute('aria-label', `${c.symbol} is campaigning for a place in the Sui Trenches Index. ${standing}. Vote on sti.boombots.fun (opens in a new tab)`);
    const lane = el('div', 'lane');
    const fill = el('i');
    fill.style.width = `${c.rank && c.leader ? Math.max(3, (c.points / c.leader) * 100) : 0}%`;
    lane.append(fill);
    const go = el('span', 'go');
    go.append(el('span', 'w', 'Vote '), 'today ↗');
    if (strip) {
      const t = el('span', 't');
      t.append(el('b', '', c.symbol), 'is campaigning for the ', el('span', 'rest', 'Sui Trenches '), 'Index');
      root.append(logo(c), t);
      if (c.rank) root.append(el('span', 'pct num', `#${c.rank}`));
      root.append(go, lane);
    } else {
      const head = el('div', 'head');
      const who = el('div', 'who');
      who.append(el('div', 'sym', c.symbol), el('div', 'is', 'is campaigning for the Sui Trenches Index'));
      const s = el('div', 'sti'); const i = el('img'); i.src = '/icon.svg'; i.alt = ''; s.append(i, el('span', '', 'STI'));
      head.append(logo(c), who, s);
      const tick = el('div', 'tick flat');
      const label = el('span');
      label.append(standing);
      tick.append(label);
      tick.style.height = '26px';
      const facts = el('div', 'facts');
      const fact = (v, k) => { const d = el('div'); d.append(el('div', 'v num', v), el('div', 'k', k)); return d; };
      facts.append(fact(c.points.toLocaleString(), 'points, last 7 days'), fact(String(c.votersToday), 'voters today'), fact(String(c.supporters), 'supporters'));
      const foot = el('div', 'foot');
      foot.append(el('p', '', 'One free vote a day. No gas.'), go);
      root.append(head, lane, tick, facts, foot);
    }
    root.hidden = false;
    document.documentElement.classList.toggle('intro', !painted);
    painted = true;
    fit(); tell();
  }

  /** A coin outside the basket: is it campaigning? */
  async function campaignFor(feed) {
    if (!want) return false;
    const list = await fetch(`${CAMPAIGNS}/campaigns`).then((r) => (r.ok ? r.json() : null)).catch(() => null);
    const c = list && list.campaigns.find((x) => x.key.toLowerCase() === want.toLowerCase() || norm(x.coin) === norm(want));
    if (!c || c.stage === 'included') return false;
    c.icon = c.icon ? `${CAMPAIGNS}/icon/${encodeURIComponent(c.key)}` : null;
    renderCampaign(feed, c, list.campaigns);
    return true;
  }

  let painted = false;
  function render(feed) {
    const coin = feed && feed.coins.find((c) => c.key === key(want) || norm(c.type) === norm(want));
    const member = coin && !coin.retiring;
    root.replaceChildren();
    root.className = `${strip ? 'strip' : 'card'}${member ? '' : ' plain'}`;
    root.href = member ? `${SITE}/?from=${encodeURIComponent(coin.key)}` : `${SITE}/`;
    const share = member ? (coin.share ?? coin.targetBps / 10_000) : 0;
    root.setAttribute('aria-label', member
      ? `${coin.symbol} is ${percent(share)} of the Sui Trenches Index. Get STI on sti.boombots.fun (opens in a new tab)`
      : 'The Sui Trenches Index. Get STI on sti.boombots.fun (opens in a new tab)');

    const mark = () => { const s = el('div', 'sti'); const i = el('img'); i.src = '/icon.svg'; i.alt = ''; s.append(i, el('span', '', 'STI')); return s; };

    if (strip) {
      const t = el('span', 't');
      if (member) { t.append(el('b', '', coin.symbol), 'is in the ', el('span', 'rest', 'Sui Trenches '), 'Index'); root.append(logo(coin)); }
      else { const i = el('img', 'logo'); i.src = '/icon.svg'; i.alt = ''; i.style.boxShadow = 'none'; t.append(el('b', '', 'STI'), 'the Sui Trenches Index'); root.append(i); }
      root.append(t);
      if (member) root.append(el('span', 'pct num', percent(share)));
      const go = el('span', 'go');
      go.append(el('span', 'w', 'Get '), 'STI ↗');
      root.append(go);
      if (feed) root.append(core(feed, member ? coin : null));
    } else if (member) {
      const head = el('div', 'head');
      const who = el('div', 'who');
      who.append(el('div', 'sym', coin.symbol), el('div', 'is', 'is in the Sui Trenches Index'));
      head.append(logo(coin), who, mark());
      const tick = el('div', 'tick');
      const label = el('span');
      label.append(el('b', 'num', percent(share)), ' of the basket');
      tick.append(label);
      const facts = el('div', 'facts');
      const fact = (v, k) => { const d = el('div'); d.append(el('div', 'v num', v), el('div', 'k', k)); return d; };
      facts.append(fact(compact(coin.held, coin.decimals), `${coin.symbol} in the vault`));
      if (feed.holders) facts.append(fact(String(feed.holders), 'STI holders'));
      facts.append(fact(coin.since ? new Date(coin.since).toLocaleDateString('en', { month: 'short', year: 'numeric' }) : 'Day one', 'member since'));
      const foot = el('div', 'foot');
      foot.append(el('p', '', `Every STI minted buys ${coin.symbol}.`), el('span', 'go', 'Get STI ↗'));
      root.append(head, core(feed, coin), tick, facts, foot);
    } else {
      const head = el('div', 'head');
      const i = el('img', 'logo'); i.src = '/icon.svg'; i.alt = ''; i.style.boxShadow = 'none'; i.style.background = 'none';
      const who = el('div', 'who');
      who.append(el('div', 'sym', 'STI'), el('div', 'is', 'Sui Trenches Index'));
      head.append(i, who);
      const foot = el('div', 'foot');
      foot.append(el('p', '', feed?.holders ? `${feed.holders} holders` : ''), el('span', 'go', 'Get STI ↗'));
      root.append(head);
      if (feed) root.append(core(feed, null));
      root.append(el('p', 'lede', feed ? `One token that holds ${feed.coins.length} Sui small caps, run by its holders.` : 'One token that holds a basket of Sui small caps, run by its holders.'), foot);
    }
    root.hidden = false;
    document.documentElement.classList.toggle('intro', !painted);   // the core is drilled once, not on every refresh
    painted = true;
    fit(); place(); tell();
  }

  addEventListener('resize', () => { fit(); place(); tell(); });
  document.fonts?.ready.then(() => { fit(); place(); tell(); });

  // the last good feed paints at once; the fresh one replaces it
  let shown = false;
  try {
    const kept = JSON.parse(localStorage.getItem('sti-badge') || 'null');
    // only a member's badge paints from the cache: a campaign's standing has to be fresh
    if (kept?.coins?.some((c) => c.key === key(want) || norm(c.type) === norm(want))) { render(kept); shown = true; }
  } catch { /* no storage here */ }
  fetch(`${FEED}/badge`).then((r) => (r.ok ? r.json() : Promise.reject(new Error(r.status)))).then(async (feed) => {
    try { localStorage.setItem('sti-badge', JSON.stringify(feed)); } catch { /* fine */ }
    const member = feed.coins.some((c) => (c.key === key(want) || norm(c.type) === norm(want)) && !c.retiring);
    if (member || !(await campaignFor(feed))) render(feed);
  }).catch(() => { if (!shown) render(null); });
})();
</script>
