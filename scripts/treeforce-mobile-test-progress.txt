function createTreeforceTestStore(wallet,storage=globalThis.localStorage){
 if(typeof wallet!=='string'||!/^0x[0-9a-f]{1,64}$/i.test(wallet))throw Error('Connect your NFTree wallet to save.');
 const key='treeforce89-mobile-test-progress-v1:'+wallet.slice(2).toLowerCase().padStart(64,'0');
 const count=value=>Number.isSafeInteger(value)&&value>=0?value:0;
 const run=value=>{if(!value||typeof value.runId!=='string'||!/^grove-[a-f0-9]{32}$/.test(value.runId))throw Error('This mission report could not be saved.');const out={runId:value.runId,ranked:value.ranked!==false,cleared:value.cleared===true};for(const field of ['score','wave','kills','perfect','maxStage','shotsFired','shotsHit','maxChain','clearedWaves','cleanWaves','turboPickups','boomPickups'])out[field]=count(value[field]);out.wave=Math.min(9,out.wave);out.maxStage=Math.min(4,out.maxStage);return out};
 const empty=()=>({version:1,username:'',bestScore:0,bestWave:0,badges:[],runs:[],pending:null});
 const read=()=>{const text=storage.getItem(key);if(!text)return empty();let value;try{value=JSON.parse(text)}catch{throw Error('Saved test data could not be read. Your current result is still on this screen.')}if(value?.version!==1)throw Error('Saved test data has an unsupported version.');return{version:1,username:typeof value.username==='string'?value.username.slice(0,20):'',bestScore:count(value.bestScore),bestWave:Math.min(10,count(value.bestWave)),badges:Array.isArray(value.badges)?value.badges.filter(id=>mh.some(b=>b[0]===id)):[],runs:Array.isArray(value.runs)?value.runs.slice(0,50).map(run):[],pending:value.pending?run(value.pending):null}};
 const write=value=>{storage.setItem(key,JSON.stringify(value));return value};
 return{read,queue(value){const previous=read();return write({...previous,pending:run(value)})},save(username,value){username=String(username).trim();if(!/^[A-Za-z0-9][A-Za-z0-9 _-]{2,19}$/.test(username))throw Error('Use 3–20 letters, numbers, spaces, underscores or hyphens.');const previous=read(),current=value?run(value):null;if(current?.ranked===false)throw Error('Practice runs do not earn saved scores or achievements.');const runs=current?[current,...previous.runs.filter(r=>r.runId!==current.runId)].slice(0,50):previous.runs;const badges=[...new Set([...previous.badges,...current?hh(current):[]])];return write({...previous,username,runs,badges,bestScore:Math.max(previous.bestScore,current?.score||0),bestWave:Math.max(previous.bestWave,current?current.wave+1:0),pending:current?null:previous.pending})}};
}
function _h({session}){
 const E=(tag,props,...children)=>(0,x.createElement)(tag,props,...children);
 const [name,setName]=x.useState(''),[profile,setProfile]=x.useState(null),[pending,setPending]=x.useState(null),[message,setMessage]=x.useState(''),[entries,setEntries]=x.useState([]),[boardError,setBoardError]=x.useState(false),[busy,setBusy]=x.useState(false);const card=x.useRef(null);
 const refresh=async()=>{try{const response=await fetch(gh);if(!response.ok)throw Error();const data=await response.json();setEntries(data.entries||[]);setBoardError(false)}catch{setBoardError(true)}};
 x.useEffect(()=>{refresh()},[]);
 x.useEffect(()=>{setPending(null);setProfile(null);setName('');setMessage('');if(!session)return;try{const saved=createTreeforceTestStore(session.wallet).read();setProfile(saved);setName(saved.username);setPending(saved.pending);if(saved.pending)setMessage('Your last test result is ready to save.')}catch{setMessage('Device storage is unavailable. You can try saving your result after the mission.')}},[session?.wallet]);
 x.useEffect(()=>{const ended=event=>{const result=event.detail;if(!session||result?.runId!==session.runId)return;if(result.ranked===false){setPending(null);setMessage('Practice run finished. Scores and achievements are not saved for practice runs.');return}setPending(result);setMessage('Mission finished. Enter your pilot username and save this result.');try{setProfile(createTreeforceTestStore(session.wallet).queue(result))}catch{setMessage('Your result is ready. Device storage is unavailable; keep this page open and retry saving.')}};window.addEventListener('treeforce-run-ended',ended);return()=>window.removeEventListener('treeforce-run-ended',ended)},[session?.wallet,session?.runId]);
 x.useEffect(()=>{if(!pending)return;const id=requestAnimationFrame(()=>card.current?.scrollIntoView({behavior:'smooth',block:'start'}));return()=>cancelAnimationFrame(id)},[pending?.runId]);
 const save=event=>{event.preventDefault();setBusy(true);try{const saved=createTreeforceTestStore(session.wallet).save(name,pending);setProfile(saved);setName(saved.username);setPending(null);setMessage('Saved on this device for '+saved.username+'. '+(pending?'Score and achievements saved.':'Pilot username saved.'))}catch(error){setMessage('Not saved. '+(error instanceof Error?error.message:'Device storage is unavailable. Please retry.'))}finally{setBusy(false)}};
 const badges=new Set(profile?.badges||[]);const unlocked=new Set(pending?.ranked!==false&&pending?hh(pending):[]);
 return E(S.Fragment,null,
 session&&E('details',{className:'pilot-board mobile-test-profile',open:!!pending||!!message,ref:card},E('summary',null,pending?'Save your mission result':'Your test pilot & saved results'),
 E('div',{className:'pilot-heading'},E('span',null,'MOBILE TEST PILOT'),E('h2',null,pending?'Mission report · '+Number(pending.score||0).toLocaleString()+' points':'Your saved test results'),E('p',null,'Saved only in this browser, for this wallet. Separate from live rankings.')),
 E('form',{onSubmit:save},E('label',{htmlFor:'mobile-test-pilot-name'},'Pilot username'),E('div',{className:'pilot-name-row'},E('input',{id:'mobile-test-pilot-name',value:name,onChange:event=>setName(event.target.value),minLength:3,maxLength:20,pattern:'[A-Za-z0-9](?:[A-Za-z0-9 _]|-){2,19}',required:true,autoComplete:'nickname',placeholder:'Choose your pilot name',onKeyDown:event=>event.stopPropagation(),onKeyUp:event=>event.stopPropagation()}),E('button',{type:'submit',disabled:busy||pending?.ranked===false},busy?'Saving…':pending?'Save result':'Save username')),
 pending&&E('p',null,'Wave '+(Number(pending.wave||0)+1)+' · '+(pending.cleared?'Canopy secured':'Canopy breach')+' · '+unlocked.size+' achievements this run'),E('p',{role:'status','aria-live':'polite',className:'mobile-save-status'},message)),
 E('p',{className:'mobile-test-record'},'Best score: '+Number(profile?.bestScore||0).toLocaleString()+' · Furthest wave: '+(profile?.bestWave||0)+' · Saved runs: '+(profile?.runs?.length||0)),
 E('small',null,'Test progress stays on this device. Clearing browser data removes it. Payments remain disabled.'),
 E('details',null,E('summary',null,'Test achievements · '+badges.size+' / '+mh.length),E('div',{className:'pilot-badges'},...mh.map(([id,title,description,,,icon])=>E('article',{key:id,className:badges.has(id)?'earned':''},E('span',{className:'pilot-medal','aria-hidden':true},icon),E('div',null,E('h3',null,title),E('p',null,description),E('small',null,badges.has(id)?'SAVED ON THIS DEVICE':unlocked.has(id)?'UNLOCKED THIS RUN · SAVE RESULT':'LOCKED')))))),
 profile?.runs?.length>0&&E('details',null,E('summary',null,'Recent saved test runs'),E('ul',null,...profile.runs.slice(0,5).map(result=>E('li',{key:result.runId},Number(result.score).toLocaleString()+' points · Wave '+(result.wave+1)+(result.cleared?' · Completed':'')))))),
 E('details',{className:'pilot-board mobile-test-board','aria-label':'Live pilot leaderboard'},E('summary',null,'View live leaderboard'),E('div',{className:'pilot-heading'},E('span',null,'CANOPY HALL OF FAME'),E('h2',null,'Live Pilot Leaderboard'),E('p',null,'Read-only. Mobile test results do not enter these rankings.')),
 E('div',{className:'pilot-table'},E('table',null,E('caption',null,'Top 50 pilots · production scores'),E('thead',null,E('tr',null,...['Rank','Pilot','Score','Badges'].map(title=>E('th',{key:title,scope:'col'},title)))),E('tbody',null,...entries.map((entry,index)=>E('tr',{key:index},E('td',null,index+1),E('th',{scope:'row'},entry.username),E('td',null,Number(entry.score).toLocaleString()),E('td',null,(entry.badges||[]).length+' / '+mh.length)))))),!entries.length&&E('p',null,boardError?'Leaderboard could not load.':'No scores yet.'),E('button',{className:'pilot-refresh',onClick:refresh},'Refresh leaderboard')));
}
